"""    # Check input validity    if len(bins) != len(edges) - 1:        raise ValueError("Invalid input: bins and edges must have the same length")    # Initialize integral    integral = 0    # Loop over bins    for i in range(len(bins)):        # Compute bin width        width = edges[i + 1] - edges[i]        # Compute bin height        height = bins[i]        # Compute bin area        area = width * height        # Add area to integral        integral += area    return integral

def difference(d1, d2, level=-1):    """Return a dictionary with items from *d1* not contained in *d2*.    *level* sets the maximum depth of recursion. For infinite recursion,    set that to -1. For level 3,    if a key is present both in *d1* and *d2* but has different values,    it is included into the difference.    See :func:`intersection` for more details.    *d1* and *d2* remain unchanged. However, *d1* or some of its    subdictionaries may be returned directly.    Make a deep copy of the result when appropriate.    .. versionadded:: 0.5       add keyword argument *level*.    """    if level == -1:        return d1    else:        result = {}        for key, value in d1.items():            if key not in d2 or d2[key] != value:                result[key] = value        for key, value in d2.items():            if key not in d1 or d1[key] != value:                result[key] = value        return result
def fill(self, coord, weight=1):    """Fill histogram at *coord* with the given *weight*.    Coordinates outside the histogram edges are ignored.    """    if not isinstance(coord, tuple) or len(coord) != 2:        raise ValueError("Coordinate must be a tuple of length 2")    if not isinstance(weight, (int, float)):        raise ValueError("Weight must be a number")    if weight <= 0:        raise ValueError("Weight must be positive")    x, y = coord    if x < 0 or x >= self.width or y < 0 or y >= self.height:        return    self.data[x][y] += weight

def format(self, sql: AnyStr, params: Union[Dict[Union[str, int], Any], Sequence[Any]]) -> Tuple[AnyStr, Union[Dict[Union[str, int], Any], Sequence[Any]]]:    """    Convert the SQL query to use the out-style parameters instead of    the in-style parameters.    *sql* (:class:`str` or :class:`bytes`) is the SQL query.    *params* (:class:`~collections.abc.Mapping` or :class:`~collections.abc.Sequence`)    contains the set of in-style parameters. It maps each parameter    (:class:`str` or :class:`int`) to value. If :attr:`.SQLParams.in_style`    is a named parameter style. then *params* must be a :class:`~collections.abc.Mapping`.    If :attr:`.SQLParams.in_style` is an ordinal parameter style, then    *params* must be a :class:`~collections.abc.Sequence`.    Returns a :class:`tuple` containing:    -	The formatted SQL query (:class:`str` or :class:`bytes`).    -	The set of converted out-style parameters (:class:`dict` or        :class:`list`).    """    if not isinstance(sql, (str, bytes)):        raise TypeError("sql must be a str or bytes object")    if not isinstance(params, (dict, list)):        raise TypeError("params must be a dict or list object")    if isinstance(params, dict):        # Named parameter style        if not all(isinstance(key, str) for key in params.keys()):            raise ValueError("keys of params must be str")        if not all(isinstance(value, (str, int)) for value in params.values()):            raise ValueError("values of params must be str or int")        # Convert named parameters to ordinal parameters        ordinal_params = []        for key, value in params.items():            ordinal_params.append(value)        # Replace named parameters with ordinal parameters in sql        for i, param in enumerate(ordinal_params):            sql = sql.replace(":" + key, "?", 1)        return sql, ordinal_params    else:        # Ordinal parameter style        if not all(isinstance(param, (str, int)) for param in params):            raise ValueError("params must be a list of str or int")        # Replace ordinal parameters with named parameters in sql        for i, param in enumerate(params):            sql = sql.replace("?", ":" + str(i), 1)        return sql, params
def formatmany(self, sql: AnyStr, many_params: Union[Iterable[Dict[Union[str, int], Any]], Iterable[Sequence[Any]]]) -> Tuple[AnyStr, Union[List[Dict[Union[str, int], Any]], List[Sequence[Any]]]]:    """    Convert the SQL query to use the out-style parameters instead of the    in-style parameters.    *sql* (:class:`str` or :class:`bytes`) is the SQL query.    *many_params* (:class:`~collections.abc.Iterable`) contains each set    of in-style parameters (*params*).    -   *params* (:class:`~collections.abc.Mapping` or :class:`~collections.abc.Sequence`)        contains the set of in-style parameters. It maps each parameter        (:class:`str` or :class:`int`) to value. If :attr:`.SQLParams.in_style`        is a named parameter style. then *params* must be a :class:`~collections.abc.Mapping`.        If :attr:`.SQLParams.in_style` is an ordinal parameter style. then        *params* must be a :class:`~collections.abc.Sequence`.    Returns a :class:`tuple` containing:    -   The formatted SQL query (:class:`str` or :class:`bytes`).    -   A :class:`list` containing each set of converted out-style        parameters (:class:`dict` or :class:`list`).    """    if not isinstance(sql, (str, bytes)):        raise TypeError("sql must be a str or bytes")    if not isinstance(many_params, Iterable):        raise TypeError("many_params must be an Iterable")    out_params = []    for params in many_params:        if isinstance(params, Mapping):            out_params.append({key: value for key, value in params.items() if value is not None})        elif isinstance(params, Sequence):            out_params.append([value for value in params if value is not None])        else:            raise TypeError("params must be a Mapping or Sequence")    return sql, out_params

def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    result = []    for expr in args:        if expr.op == '&':            if expr.left.op == '|' and expr.left.right == expr.right:                result.append(expr.left.left)            elif expr.right.op == '|' and expr.right.right == expr.left:                result.append(expr.right.left)        elif expr.op == '|':            if expr.left.op == '&' and expr.left.right == expr.right:                result.append(expr.left.left)            elif expr.right.op == '&' and expr.right.right == expr.left:                result.append(expr.right.left)        else:            result.append(expr)    return result

def _fromutc(self, dt):    """    Given a timezone-aware datetime in a given timezone, calculates a    timezone-aware datetime in a new timezone.    Since this is the one time that we *know* we have an unambiguous    datetime object, we take this opportunity to determine whether the    datetime is ambiguous and in a "fold" state (e.g. if it's the first    occurrence, chronologically, of the ambiguous datetime).    :param dt:        A timezone-aware :class:`datetime.datetime` object.    """    if not isinstance(dt, datetime.datetime):        raise TypeError("dt must be a datetime.datetime object")    if dt.tzinfo is None:        raise ValueError("dt must be a timezone-aware datetime object")    # Calculate the new datetime in the new timezone    new_dt = dt.astimezone(self.tzinfo)    # Determine whether the datetime is ambiguous and in a "fold" state    is_ambiguous = self.is_ambiguous(dt)    is_fold = self.is_fold(dt)    return new_dt, is_ambiguous, is_fold
def normalized(self):    """    Return a version of this object represented entirely using integer    values for the relative attributes.    >>> relativedelta(days=1.5, hours=2).normalized()    relativedelta(days=+1, hours=+14)    :return:        Returns a :class:`dateutil.relativedelta.relativedelta` object.    """    # Step 1: Calculate the total number of seconds in the relativedelta    total_seconds = (self.days * 24 * 60 * 60) + (self.hours * 60 * 60) + (self.minutes * 60) + self.seconds    # Step 2: Calculate the number of days, hours, minutes, and seconds in the total number of seconds    days = total_seconds // (24 * 60 * 60)    hours = (total_seconds - (days * 24 * 60 * 60)) // (60 * 60)    minutes = (total_seconds - (days * 24 * 60 * 60) - (hours * 60 * 60)) // 60    seconds = total_seconds - (days * 24 * 60 * 60) - (hours * 60 * 60) - (minutes * 60)    # Step 3: Return a new relativedelta object with the calculated values    return relativedelta(days=days, hours=hours, minutes=minutes, seconds=seconds)

def next_version(version):    """Next version identifier following existing pattern.    Must deal with both zero-prefixed and non-zero prefixed versions.    """    # Check if the input is a valid version string    if not re.match(r"^\d+\.\d+\.\d+$", version):        raise ValueError("Invalid version string")    # Split the version string into major, minor, and patch components    major, minor, patch = version.split(".")    # Increment the patch version by 1    patch = str(int(patch) + 1)    # If the patch version is 10 or higher, increment the minor version by 1 and reset the patch version to 0    if int(patch) >= 10:        minor = str(int(minor) + 1)        patch = "0"    # If the minor version is 10 or higher, increment the major version by 1 and reset the minor and patch versions to 0    if int(minor) >= 10:        major = str(int(major) + 1)        minor = "0"        patch = "0"    # Return the next version string    return ".".join([major, minor, patch])
def validate_version_inventories(self, version_dirs):    """    Validate that each version has an inventory up to that point and keep a record of any content digests    different from those in the root inventory so that we can also check them when validating the content.    version_dirs is an array of version directory names and is assumed to be in version sequence (1, 2, 3...).    """    # Initialize variables    root_inventory = None    version_inventories = {}    content_digests = set()    # Iterate over version directories    for version_dir in version_dirs:        # Get the inventory for the current version        inventory = self.get_inventory(version_dir)        # If the inventory is not found, raise an error        if inventory is None:            raise ValueError(f"Inventory not found for version {version_dir}")        # If the inventory is not a dictionary, raise an error        if not isinstance(inventory, dict):            raise ValueError(f"Inventory for version {version_dir} is not a dictionary")        # If the inventory is empty, raise an error        if not inventory:            raise ValueError(f"Inventory for version {version_dir} is empty")        # If the root inventory has not been set, set it to the current inventory        if root_inventory is None:            root_inventory = inventory        # Add the current inventory to the version inventories dictionary        version_inventories[version_dir] = inventory        # Add the content digests from the current inventory to the content digests set        content_digests.update(inventory.keys())    # If the root inventory is not found, raise an error    if root_inventory is None:        raise ValueError("Root inventory not found")    # If the root inventory is not a dictionary, raise an error    if not isinstance(root_inventory, dict):        raise ValueError("Root inventory is not a dictionary")    # If the root inventory is empty, raise an error    if not root_inventory:        raise ValueError("Root inventory is empty")    # Check that the root inventory contains all the content digests in the content digests set    for digest in content_digests:        if digest not in root_inventory:            raise ValueError(f"Content digest {digest} not found in root inventory")    # Return the version inventories dictionary    return version_inventories
def find_path_type(path):    """Return a string indicating the type of thing at the given path.    Return values:        'root' - looks like an OCFL Storage Root        'object' - looks like an OCFL Object        'file' - a file, might be an inventory        other string explains error description    Looks only at "0=*" Namaste files to determine the directory type.    """    try:        namaste_files = [f for f in os.listdir(path) if f.startswith("0=*")]        if len(namaste_files) == 0:            return "file"        elif len(namaste_files) == 1:            if namaste_files[0].endswith("inventory.json"):                return "object"            else:                return "root"        else:            return "other"    except OSError as e:        return "other"

def deep_merge_nodes(nodes):    """    Given a nested borgmatic configuration data structure as a list of tuples in the form of:        (            ruamel.yaml.nodes.ScalarNode as a key,            ruamel.yaml.nodes.MappingNode or other Node as a value,        ),    ... deep merge any node values corresponding to duplicate keys and return the result. If    there are colliding keys with non-MappingNode values (e.g., integers or strings), the last    of the values wins.    For instance, given node values of:        [            (                ScalarNode(tag='tag:yaml.org,2002:str', value='retention'),                MappingNode(tag='tag:yaml.org,2002:map', value=[                    (                        ScalarNode(tag='tag:yaml.org,2002:str', value='keep_hourly'),                        ScalarNode(tag='tag:yaml.org,2002:int', value='24')                    ),                    (                        ScalarNode(tag='tag:yaml.org,2002:str', value='keep_daily'),                        ScalarNode(tag='tag:yaml.org,2002:int', value='7')                    ),                ]),            ),            (                ScalarNode(tag='tag:yaml.org,2002:str', value='retention'),                MappingNode(tag='tag:yaml.org,2002:map', value=[                    (                        ScalarNode(tag='tag:yaml.org,2002:str', value='keep_daily'),                        ScalarNode(tag='tag:yaml.org,2002:int', value='5')                    ),                ]),            ),        ]    ... the returned result would be:        [            (                ScalarNode(tag='tag:yaml.org,2002:str', value='retention'),                MappingNode(tag='tag:yaml.org,2002:map', value=[                    (                        ScalarNode(tag='tag:yaml.org,2002:str', value='keep_hourly'),                        ScalarNode(tag='tag:yaml.org,2002:int', value='24')                    ),                    (                        ScalarNode(tag='tag:yaml.org,2002:str', value='keep_daily'),                        ScalarNode(tag='tag:yaml.org,2002:int', value='5')                    ),                ]),            ),        ]    The purpose of deep merging like this is to support, for instance, merging one borgmatic    configuration file into another for reuse, such that a configuration section ("retention",    etc.) does not completely replace the corresponding section in a merged file.    """    # Step 1: Create a dictionary to store the merged nodes    merged_nodes = {}    # Step 2: Iterate over the input nodes and merge them    for node in nodes:        # Step 2.1: Get the key and value from the current node        key, value = node        # Step 2.2: Check if the key is already in the merged nodes dictionary        if key in merged_nodes:            # Step 2.3: If the key is already in the dictionary, merge the values            merged_nodes[key] = deep_merge_nodes(merged_nodes[key], value)        else:            # Step 2.4: If the key is not in the dictionary, add the node to the dictionary            merged_nodes[key] = value    # Step 3: Return the merged nodes    return merged_nodes.items()

def deep_merge_nodes(nodes):    """    Given a nested borgmatic configuration data structure as a list of tuples in the form of:        (            ruamel.yaml.nodes.ScalarNode as a key,            ruamel.yaml.nodes.MappingNode or other Node as a value,        ),    ... deep merge any node values corresponding to duplicate keys and return the result. def subprocess_run_helper(func, *args, timeout, extra_env=None):    """    Run a function in a sub-process.    Parameters    ----------    func : function        The function to be run.  It must be in a module that is importable.    *args : str        Any additional command line arguments to be passed in        the first argument to ``subprocess.run``.    extra_env : dict[str, str]        Any additional environment variables to be set for the subprocess.    """    import subprocess    # Check if the function is importable    if not callable(func):        raise ValueError("The function must be importable.")    # Check if the function has the correct signature    if not hasattr(func, "__call__"):        raise ValueError("The function must have a callable signature.")    # Check if the function has the correct number of arguments    if len(args) != len(func.__code__.co_varnames):        raise ValueError("The function must have the correct number of arguments.")    # Check if the function has the correct argument types    for i, arg in enumerate(args):        if not isinstance(arg, str):            raise ValueError(f"Argument {i} must be a string.")    # Check if the function has the correct return type    if not isinstance(func.__code__.co_consts[0], str):        raise ValueError("The function must have a string return type.")    # Check if the timeout is valid    if timeout is not None and not isinstance(timeout, (int, float)):        raise ValueError("The timeout must be a number or None.")    # Check if the extra_env is valid    if extra_env is not None and not isinstance(extra_env, dict):        raise ValueError("The extra_env must be a dictionary or None.")    # Run the function in a sub-process    try:        result = subprocess.run(            [sys.executable, "-c", f"import {func.__module__}; {func.__name__}({', '.join(args)})"],            env={**os.environ, **(extra_env or {})},            timeout=timeout,            check=True,            stdout=subprocess.PIPE,            stderr=subprocess.PIPE,        )    except subprocess.CalledProcessError as e:        raise RuntimeError(f"The function failed with exit code {e.returncode}.") from e    except subprocess.TimeoutExpired as e:        raise RuntimeError(f"The function timed out after {timeout} seconds.") from e    # Return the result    return result.stdout.decode().strip()

def scale(self, other=None):    """Get or set the scale of the graph.    If *other* is ``None``, return the scale of this graph.    If a numeric *other* is provided, rescale to that value.    If the graph has unknown or zero scale,    rescaling that will raise :exc:`~.LenaValueError`.    To get meaningful results, graph's fields are used.    Only the last coordinate is rescaled.    For example, if the graph has *x* and *y* coordinates,    then *y* will be rescaled, and for a 3-dimensional graph    *z* will be rescaled.    All errors are rescaled together with their coordinate.    """    if other is None:        return self._scale    else:        if not isinstance(other, numbers.Number):            raise TypeError("Scale must be a number")        if other <= 0:            raise ValueError("Scale must be positive")        if self._scale is None:            raise LenaValueError("Cannot rescale graph with unknown scale")        if self._scale == 0:            raise LenaValueError("Cannot rescale graph with zero scale")        self._scale = other        for field in self._fields:            field.rescale(other)        for error in self._errors:            error.rescale(other)        return self
def hist_to_graph(hist, make_value=None, get_coordinate="left", field_names=("x", "y"), scale=None):    """Convert a :class:`.histogram` to a :class:`.graph`.    *make_value* is a function to set the value of a graph's point.    By default it is bin content.    *make_value* accepts a single value (bin content) without context.    This option could be used to create graph's error bars.    For example, to create a graph with errors    from a histogram where bins contain    a named tuple with fields *mean*, *mean_error* and a context    one could use    >>> make_value = lambda bin_: (bin_.mean, bin_.mean_error)    *get_coordinate* defines what the coordinate    of a graph point created from a histogram bin will be.    It can be "left" (default), "right" and "middle".    *field_names* set field names of the graph. Their number    must be the same as the dimension of the result.    For a *make_value* above they would be    *("x", "y_mean", "y_mean_error")*.    *scale* becomes the graph's scale (unknown by default).    If it is ``True``, it uses the histogram scale.    *hist* must contain only numeric bins (without context)    or *make_value* must remove context when creating a numeric graph.    Return the resulting graph.    """    # Validate input    if not isinstance(hist, Histogram):        raise TypeError("hist must be a Histogram")    if not callable(make_value):        raise TypeError("make_value must be a callable")    if get_coordinate not in ("left", "right", "middle"):        raise ValueError("get_coordinate must be one of 'left', 'right', or 'middle'")    if not isinstance(field_names, tuple):        raise TypeError("field_names must be a tuple")    if len(field_names) != 2:        raise ValueError("field_names must have 2 elements")    if scale is not None and not isinstance(scale, bool):        raise TypeError("scale must be a boolean")    # Create a new graph    graph = Graph()    # Set the graph's scale    if scale is True:        graph.scale = hist.scale    # Set the graph's field names    graph.field_names = field_names    # Iterate over the histogram bins    for bin_ in hist.bins:        # Get the bin's coordinate        if get_coordinate == "left":            coordinate = bin_.left        elif get_coordinate == "right":            coordinate = bin_.right        else:            coordinate = (bin_.left + bin_.right) / 2        # Get the bin's value        value = make_value(bin_)        # Add the point to the graph        graph.add_point(coordinate, value)    return graph

def determineMetaclass(bases, explicit_mc=None):    """Determine metaclass from 1+ bases and optional explicit __metaclass__"""    if explicit_mc is not None:        return explicit_mc    metaclasses = set()    for base in bases:        if hasattr(base, '__class__'):            metaclasses.add(base.__class__)    if len(metaclasses) == 1:        return metaclasses.pop()    else:        raise TypeError("Inconsistent metaclasses found")
def pop(self, key, default=__marker):    """    D.pop(k[,d]) -> v, remove specified key and return the corresponding value. def isoparse(dt_str):    """    Parse an ISO-8601 datetime string into a :class:`datetime.datetime`.    An ISO-8601 datetime string consists of a date portion, followed    optionally by a time portion - the date and time portions are separated    by a single character separator, which is ``T`` in the official    standard. Incomplete date formats (such as ``YYYY-MM``) may *not* be    combined with a time portion.    Supported date formats are:    Common:    - ``YYYY``    - ``YYYY-MM`` or ``YYYYMM``    - ``YYYY-MM-DD`` or ``YYYYMMDD``    Uncommon:    - ``YYYY-Www`` or ``YYYYWww`` - ISO week (day defaults to 0)    - ``YYYY-Www-D`` or ``YYYYWwwD`` - ISO week and day    The ISO week and day numbering follows the same logic as    :func:`datetime.date.isocalendar`.    Supported time formats are:    - ``hh``    - ``hh:mm`` or ``hhmm``    - ``hh:mm:ss`` or ``hhmmss``    - ``hh:mm:ss.ssssss`` (Up to 6 sub-second digits)    Midnight is a special case for `hh`, as the standard supports both    00:00 and 24:00 as a representation. The decimal separator can be    either a dot or a comma.    .. caution::        Support for fractional components other than seconds is part of the        ISO-8601 standard, but is not currently implemented in this parser.    Supported time zone offset formats are:    - `Z` (UTC)    - `±HH:MM`    - `±HHMM`    - `±HH`    Offsets will be represented as :class:`dateutil.tz.tzoffset` objects,    with the exception of UTC, which will be represented as    :class:`dateutil.tz.tzutc`. Time zone offsets equivalent to UTC (such    as `+00:00`) will also be represented as :class:`dateutil.tz.tzutc`.    :param dt_str:        A string or stream containing only an ISO-8601 datetime string    :return:        Returns a :class:`datetime.datetime` representing the string.        Unspecified components default to their lowest value.    .. warning::        As of version 2.7.0, the strictness of the parser should not be        considered a stable part of the contract. Any valid ISO-8601 string        that parses correctly with the default settings will continue to        parse correctly in future versions, but invalid strings that        currently fail (e.g. ``2017-01-01T00:00+00:00:00``) are not        guaranteed to continue failing in future versions if they encode        a valid date.    .. versionadded:: 2.7.0    """    # Check if the input string is empty    if not dt_str:        raise ValueError("Invalid input: empty string")    # Check if the input string is a valid ISO-8601 datetime string    if not is_valid_iso_8601_datetime(dt_str):        raise ValueError("Invalid input: not a valid ISO-8601 datetime string")    # Parse the date portion of the input string    date_part = dt_str.split("T")[0]    year, month, day = parse_iso_8601_date(date_part)    # Parse the time portion of the input string    time_part = dt_str.split("T")[1]    hour, minute, second, microsecond = parse_iso_8601_time(time_part)    # Create a datetime object from the parsed components    dt = datetime(year, month, day, hour, minute, second, microsecond)    # Check if the input string contains a time zone offset    if "Z" in dt_str:        # If the input string contains a time zone offset,        # convert the datetime object to UTC        dt = dt.astimezone(pytz.utc)    return dt
def parse(self, timestr, default=None, ignoretz=False, tzinfos=None, **kwargs):    """    Parse the date/time string into a :class:`datetime.datetime` object.    :param timestr:        Any date/time string using the supported formats.    :param default:        The default datetime object, if this is a datetime object and not        ``None``, elements specified in ``timestr`` replace elements in the        default object.    :param ignoretz:        If set ``True``, time zones in parsed strings are ignored and a        naive :class:`datetime.datetime` object is returned.    :param tzinfos:        Additional time zone names / aliases which may be present in the        string. This argument maps time zone names (and optionally offsets        from those time zones) to time zones. This parameter can be a        dictionary with timezone aliases mapping time zone names to time        zones or a function taking two parameters (``tzname`` and        ``tzoffset``) and returning a time zone.        The timezones to which the names are mapped can be an integer        offset from UTC in seconds or a :class:`tzinfo` object.        .. doctest::           :options: +NORMALIZE_WHITESPACE            >>> from dateutil.parser import parse            >>> from dateutil.tz import gettz            >>> tzinfos = {"BRST": -7200, "CST": gettz("America/Chicago")}            >>> parse("2012-01-19 17:21:00 BRST", tzinfos=tzinfos)            datetime.datetime(2012, 1, 19, 17, 21, tzinfo=tzoffset(u'BRST', -7200))            >>> parse("2012-01-19 17:21:00 CST", tzinfos=tzinfos)            datetime.datetime(2012, 1, 19, 17, 21,                              tzinfo=tzfile('/usr/share/zoneinfo/America/Chicago'))        This parameter is ignored if ``ignoretz`` is set.    :param \\*\\*kwargs:        Keyword arguments as passed to ``_parse()``.    :return:        Returns a :class:`datetime.datetime` object or, if the        ``fuzzy_with_tokens`` option is ``True``, returns a tuple, the        first element being a :class:`datetime.datetime` object, the second        a tuple containing the fuzzy tokens.    :raises ParserError:        Raised for invalid or unknown string format, if the provided        :class:`tzinfo` is not in a valid format, or if an invalid date        would be created.    :raises TypeError:        Raised for non-string or character stream input.    :raises OverflowError:        Raised if the parsed date exceeds the largest valid C integer on        your system.    """    if not isinstance(timestr, str):        raise TypeError("Expected string, got %s" % type(timestr))    if default is not None and not isinstance(default, datetime.datetime):        raise TypeError("Expected datetime, got %s" % type(default))    if ignoretz and tzinfos is not None:        raise ValueError("Cannot use tzinfos with ignoretz")    if tzinfos is not None and not isinstance(tzinfos, (dict, callable)):        raise TypeError("Expected dict or callable, got %s" % type(tzinfos))    if not isinstance(kwargs, dict):        raise TypeError("Expected dict, got %s" % type(kwargs))    # Parse the date/time string    try:        parsed = _parse(timestr, **kwargs)    except ValueError as e:        raise ParserError(e)    # If the parsed date/time is naive, return it as is    if parsed.tzinfo is None:        return parsed    # If the parsed date/time has a timezone, convert it to the target timezone    if not ignoretz:        if tzinfos is not None:            tzinfo = tzinfos.get(parsed.tzinfo.tzname(None))            if tzinfo is not None:                parsed = parsed.astimezone(tzinfo)        else:            parsed = parsed.astimezone(datetime.timezone.utc)    # If a default datetime object was provided, update it with the parsed elements    if default is not None:        for attr in ("year", "month", "day", "hour", "minute", "second", "microsecond"):            value = getattr(parsed, attr)            if value is not None:                setattr(default, attr, value)        parsed = default    return parsed
def fromutc(self, dt):    """    Given a timezone-aware datetime in a given timezone, calculates a    timezone-aware datetime in a new timezone.    Since this is the one time that we *know* we have an unambiguous    datetime object, we take this opportunity to determine whether the    datetime is ambiguous and in a "fold" state (e.g. if it's the first    occurrence, chronologically, of the ambiguous datetime).    :param dt:        A timezone-aware :class:`datetime.datetime` object.    """    # Validate input    if not isinstance(dt, datetime.datetime):        raise TypeError("dt must be a datetime.datetime object")    if dt.tzinfo is None:        raise ValueError("dt must be a timezone-aware datetime object")    # Calculate the new datetime in the new timezone    new_dt = dt.astimezone(self.tzinfo)    # Determine if the datetime is ambiguous and in a "fold" state    if new_dt.tzinfo.is_ambiguous(new_dt):        new_dt = new_dt.replace(fold=1)    return new_dt

def validate_arg_deprecation(self, cli_args, answer_file_args):    """Validates and prints the deprecated arguments.    :param cli_args: the dict of arguments from cli    :param answer_file_args:  the dict of arguments from files    """    # Check if the argument is deprecated    for arg in cli_args:        if arg in self.deprecated_args:            print(f"Warning: The argument '{arg}' is deprecated and will be removed in a future version.")    # Check if the argument is deprecated in the answer file    for arg in answer_file_args:        if arg in self.deprecated_args:            print(f"Warning: The argument '{arg}' is deprecated and will be removed in a future version.")    # Check if the argument is deprecated in both the cli and answer file    for arg in cli_args:        if arg in answer_file_args and arg in self.deprecated_args:            print(f"Warning: The argument '{arg}' is deprecated and will be removed in a future version.")
def get_parser_option_specs(self, command_name):    """Gets all the options for the specified command    :param command_name: the command name (main, virsh, ospd, etc...)    :return: the list of all command options    """    try:        command = self.get_command(command_name)        return command.option_specs    except KeyError:        raise ValueError(f"Invalid command name: {command_name}")    except AttributeError:        raise ValueError(f"Command {command_name} does not have any options")
def get_option_spec(self, command_name, argument_name):    """Gets the specification for the specified option name.    Args:        command_name (str): The name of the command.        argument_name (str): The name of the option.    Returns:        OptionSpec: The specification for the option.    Raises:        ValueError: If the command or option name is invalid.        KeyError: If the option is not found in the command.    """    # Check if the command and option names are valid    if not self.is_valid_command(command_name):        raise ValueError(f"Invalid command name: {command_name}")    if not self.is_valid_option(argument_name):        raise ValueError(f"Invalid option name: {argument_name}")    # Get the option specification    option_spec = self.get_option_specs(command_name).get(argument_name)    if option_spec is None:        raise KeyError(f"Option not found: {argument_name}")    return option_spec
def get_silent_args(self, args):    """    Returns a list of silenced argument names.    :param args: The received arguments.    :return: list, slienced argument names    """    silent_args = []    for arg in args:        if arg.startswith("-"):            silent_args.append(arg)    return silent_args
def validate_requires_args(self, args):    """Check if all the required arguments have been provided.    Args:        args (dict): A dictionary of arguments.    Raises:        ValueError: If any of the required arguments are missing.    """    required_args = ['arg1', 'arg2', 'arg3']    missing_args = []    for arg in required_args:        if arg not in args:            missing_args.append(arg)    if missing_args:        raise ValueError(f"Missing required arguments: {', '.join(missing_args)}")
def _get_conditionally_required_args(self, command_name, options_spec, args):    """List arguments with ``required_when`` condition matched.    :param command_name: the command name.    :param options_spec:  the list of command spec options.    :param args: the received input arguments    :return: list, list of argument names with matched ``required_when`` condition    """    required_args = []    for option in options_spec:        if option.get("required_when"):            condition = option["required_when"]            if isinstance(condition, str):                condition = [condition]            for c in condition:                if c in args:                    required_args.append(option["name"])                    break    return required_args
def validate_length_args(self, args):    """Check if value of arguments is not longer than length specified.    :param args: The received arguments.    """    for arg in args:        if len(arg) > self.length:            raise ValueError(f"Argument {arg} is longer than the maximum length of {self.length}")    return True
def validate_choices_args(self, args):    """Check if value of choice arguments is one of the available choices.    :param args: The received arguments.    """    for arg in args:        if arg not in self.choices:            raise ValueError(f"Invalid choice: {arg}. Must be one of {self.choices}")    return True
def validate_min_max_args(self, args):    """Check if value of arguments is between minimum and maximum values.    :param args: The received arguments.    """    # Check if the number of arguments is valid    if len(args) != 2:        raise ValueError("Invalid number of arguments")    # Check if the arguments are of the correct type    if not isinstance(args[0], int) or not isinstance(args[1], int):        raise TypeError("Arguments must be integers")    # Check if the first argument is less than or equal to the second argument    if args[0] > args[1]:        raise ValueError("First argument must be less than or equal to second argument")    # Check if the second argument is greater than or equal to the first argument    if args[1] < args[0]:        raise ValueError("Second argument must be greater than or equal to first argument")    return True
def create_complex_argumet_type(self, subcommand, type_name, option_name, spec_option):    """Build the complex argument type    :param subcommand: the command name    :param type_name: the complex type name    :param option_name: the option name    :param spec_option: option's specifications    :return: the complex type instance    """    # Check if the subcommand is valid    if subcommand not in self.subcommands:        raise ValueError(f"Invalid subcommand: {subcommand}")    # Check if the type name is valid    if type_name not in self.types:        raise ValueError(f"Invalid type name: {type_name}")    # Check if the option name is valid    if option_name not in self.options:        raise ValueError(f"Invalid option name: {option_name}")    # Check if the spec_option is valid    if spec_option not in self.specs:        raise ValueError(f"Invalid spec_option: {spec_option}")    # Create the complex argument type    complex_type = self.types[type_name](subcommand, option_name, spec_option)    return complex_type
def get_nested_custom_and_control_args(self, args):    """Split input arguments to control nested and custom.    Controls arguments: control the IR behavior. These arguments        will not be put into the spec yml file    Nested arguments: are used by the Ansible playbooks and will be put        into the spec yml file.    Custom arguments: Custom ansible variables to be used instead of the        normal nested usage.    :param args: the collected list of args.    :return: (dict, dict): flat dicts (control_args, nested_args)    """    control_args = {}    nested_args = {}    custom_args = {}    for arg in args:        if arg.startswith("control_"):            control_args[arg] = args[arg]        elif arg.startswith("nested_"):            nested_args[arg] = args[arg]        else:            custom_args[arg] = args[arg]    return control_args, nested_args, custom_args
def merge_extra_vars(vars_dict, extra_vars=None):    """Extend ``vars_dict`` with ``extra-vars``    :param vars_dict: Dictionary to merge extra-vars into    :param extra_vars: List of extra-vars    """    if extra_vars is None:        return vars_dict    for key, value in extra_vars.items():        if key not in vars_dict:            vars_dict[key] = value        elif isinstance(vars_dict[key], list):            vars_dict[key].extend(value)        else:            vars_dict[key] = [vars_dict[key], value]    return vars_dict
def ansible_playbook(ir_workspace, ir_plugin, playbook_path, verbose=None, extra_vars=None, ansible_args=None):    """Wraps the 'ansible-playbook' CLI.    :param ir_workspace: An Infrared Workspace object represents the active workspace    :param ir_plugin: An InfraredPlugin object of the current plugin    :param playbook_path: the playbook to invoke    :param verbose: Ansible verbosity level    :param extra_vars: dict. Passed to Ansible as extra-vars    :param ansible_args: dict of ansible-playbook arguments to plumb down directly to Ansible.    """    # Step 1: Validate input parameters    if not isinstance(ir_workspace, InfraredWorkspace):        raise ValueError("ir_workspace must be an InfraredWorkspace object")    if not isinstance(ir_plugin, InfraredPlugin):        raise ValueError("ir_plugin must be an InfraredPlugin object")    if not isinstance(playbook_path, str):        raise ValueError("playbook_path must be a string")    if verbose is not None and not isinstance(verbose, int):        raise ValueError("verbose must be an integer")    if extra_vars is not None and not isinstance(extra_vars, dict):        raise ValueError("extra_vars must be a dict")    if ansible_args is not None and not isinstance(ansible_args, dict):        raise ValueError("ansible_args must be a dict")    # Step 2: Build the ansible-playbook command    command = ["ansible-playbook", playbook_path]    if verbose is not None:        command.append("-" + str(verbose))    if extra_vars is not None:        for key, value in extra_vars.items():            command.append("-e")            command.append(f"{key}={value}")    if ansible_args is not None:        for key, value in ansible_args.items():            command.append(f"--{key}")            command.append(value)    # Step 3: Run the command    try:        result = subprocess.run(command, check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)        return result.stdout.decode("utf-8")    except subprocess.CalledProcessError as e:        raise RuntimeError(f"Ansible playbook execution failed with exit code {e.returncode}") from e    except OSError as e:        raise RuntimeError("Ansible playbook execution failed") from e

def get_plugin_spec_flatten_dict(plugin_dir):    """Creates a flat dict from the plugin spec    :param plugin_dir: A path to the plugin's dir    :return: A flatten dictionary contains the plugin's properties    """    try:        with open(os.path.join(plugin_dir, "plugin.spec"), "r") as f:            spec = yaml.safe_load(f)    except FileNotFoundError:        raise ValueError(f"Plugin spec file not found in {plugin_dir}")    except yaml.YAMLError:        raise ValueError(f"Invalid YAML format in plugin spec file {plugin_dir}/plugin.spec")    flatten_dict = {}    for key, value in spec.items():        if isinstance(value, dict):            for subkey, subvalue in value.items():                flatten_dict[f"{key}.{subkey}"] = subvalue        else:            flatten_dict[key] = value    return flatten_dict
