"""    # Check if params is a mapping or a sequence    if isinstance(params, Mapping):        # If params is a mapping, convert it to a dictionary        param_dict = dict(params)    elif isinstance(params, Sequence):        # If params is a sequence, convert it to a list        param_list = list(params)    else:        raise TypeError("params must be a mapping or a sequence")    # Create a new dictionary to store the converted parameters    converted_params = {}    # Replace the in-style parameters with out-style parameters    for key, value in param_dict.items():        # If the key is a string, use it as a named parameter        if isinstance(key, str):            # If the value is a list, join the list items into a string            if isinstance(value, list):                value = ", ".join(value)            # Add the key-value pair to the converted parameters dictionary            converted_params[f"@{key}"] = value        # If the key is an integer, use it as an ordinal parameter        elif isinstance(key, int):            # If the value is a list, join the list items into a string            if isinstance(value, list):                value = ", ".join(value)            # Add the key-value pair to the converted parameters dictionary            converted_params[f"${key}"] = value        else:            raise TypeError(f"Invalid parameter: {key}={value}")    # Replace the in-style parameters in the SQL query with the out-style parameters    for key, value in converted_params.items():        sql = sql.replace(f"{{{key}}}", value)    # Return the formatted SQL query and the converted parameters    return sql, converted_params

def _fromutc(self, dt):    """    Given a timezone-aware datetime in a given timezone, calculates a    timezone-aware datetime in a new timezone.    Since this is the one time that we *know* we have an unambiguous    datetime object, we take this opportunity to determine whether the    datetime is ambiguous and in a "fold" state (e.g. if it's the first    occurrence, chronologically, of the ambiguous datetime).    :param dt:        A timezone-aware :class:`datetime.datetime` object.    """    # Get the time difference between UTC and the current timezone    utc_delta = self.get_utc_offset(dt.tzinfo.zone)    # Create a new date object with the same year, month, and day as the input datetime    # but with the time part set to midnight    new_date = date(dt.year, dt.month, dt.day)    # Get the time difference between the new date and the current datetime    date_delta = (dt - new_date).total_seconds()    # Calculate the new datetime in the target timezone    new_dt = dt + timedelta(hours=utc_delta) + timedelta(hours=date_delta)    # Check if the new datetime is ambiguous    if not self.is_ambiguous(new_dt):        # The new datetime is not ambiguous, so we don't need to worry about the "fold" state        return new_dt    # The new datetime is ambiguous, so we need to check if it's in the "fold" state    fold_start = self.get_fold_start(new_dt.year, new_dt.month, new_dt.day)    # If the new datetime is before the start of the "fold", we need to adjust the time    if new_dt < fold_start:        # The new datetime is not in the "fold" state, so we can return it as-is        return new_dt    # The new datetime is in the "fold" state, so we need to adjust the time    # Get the time difference between the new datetime and the start of the "fold"    fold_delta = (fold_start - new_dt).total_seconds()    # Calculate the new datetime in the target timezone    new_dt = dt + timedelta(hours=utc_delta) + timedelta(seconds=fold_delta)    # Return the new datetime    return new_dt

def int_to_string(number: int, alphabet: List[str], padding: Optional[int] = None) -> str:    """    Convert a number to a string, using the given alphabet.    The output has the most significant digit first.    """    if not isinstance(number, int):        raise TypeError("number must be an integer.")    if not isinstance(alphabet, list):        raise TypeError("alphabet must be a list.")    if not all(isinstance(x, str) for x in alphabet):        raise TypeError("all elements of alphabet must be strings.")    if padding is not None and not isinstance(padding, int):        raise TypeError("padding must be an integer if given.")    if number == 0:        return alphabet[0]    result = ""    while number > 0:        result = alphabet[number % len(alphabet)] + result        number //= len(alphabet)    if padding is not None:        result = "0" * (padding - len(result)) + result    return result

def merge_extra_vars(vars_dict, extra_vars=None):    """Extend ``vars_dict`` with ``extra-vars``    :param vars_dict: Dictionary to merge extra-vars into    :param extra_vars: List of extra-vars    """    if not extra_vars:        return vars_dict    for key, value in extra_vars.items():        if key in vars_dict and isinstance(vars_dict[key], dict) and isinstance(value, dict):            vars_dict[key] = {**vars_dict[key], **value}        else:            vars_dict[key] = value    return vars_dict
def ansible_playbook(ir_workspace, ir_plugin, playbook_path, verbose=None, extra_vars=None, ansible_args=None):    """Wraps the 'ansible-playbook' CLI.    :param ir_workspace: An Infrared Workspace object represents the active workspace    :param ir_plugin: An InfraredPlugin object of the current plugin    :param playbook_path: the playbook to invoke    :param verbose: Ansible verbosity level    :param extra_vars: dict. Passed to Ansible as extra-vars    :param ansible_args: dict of ansible-playbook arguments to plumb down directly to Ansible.    """    # Validate input parameters    if not isinstance(ir_workspace, ir.Workspace):        raise TypeError("ir_workspace must be an Infrared Workspace object")    if not isinstance(ir_plugin, ir.Plugin):        raise TypeError("ir_plugin must be an Infrared Plugin object")    if not isinstance(playbook_path, str):        raise TypeError("playbook_path must be a string")    if verbose is not None and not isinstance(verbose, int):        raise TypeError("verbose must be an integer")    if extra_vars is not None and not isinstance(extra_vars, dict):        raise TypeError("extra_vars must be a dict")    if ansible_args is not None and not isinstance(ansible_args, dict):        raise TypeError("ansible_args must be a dict")    # Create a new Ansible playbook runner    runner = ansible.PlaybookRunner(playbook_path)    # Set the Infrared Workspace and Plugin as runner properties    runner.set_workspace(ir_workspace)    runner.set_plugin(ir_plugin)    # Set the Ansible verbosity level    if verbose is not None:        runner.set_verbosity(verbose)    # Set the extra vars    if extra_vars is not None:        runner.set_extra_vars(extra_vars)    # Set the ansible-playbook arguments    if ansible_args is not None:        runner.set_ansible_args(ansible_args)    # Run the playbook    runner.run()
def _run_playbook(cli_args, vars_dict, ir_workspace, ir_plugin):    """Runs ansible cli with vars dict    :param vars_dict: dict, Will be passed as Ansible extra-vars    :param cli_args: the list  of command line arguments    :param ir_workspace: An Infrared Workspace object represents the active     workspace    :param ir_plugin: An InfraredPlugin object of the current plugin    :return: ansible results    """    import yaml    import infrared.core.utils as ir_utils    try:        vars_file = ir_utils.write_yaml(vars_dict, ir_workspace.config.temp_path / "ansible_vars.yml")        cli_args.extend(["--"], vars_file.as_posix())        ir_logger = ir_workspace.logger.getLogger("infrared.core.ansible")        ansible_runner = ir_plugin.get_adapter("ansible_runner")        results = ansible_runner.run(cli_args, ir_logger)        return results    except Exception as e:        ir_logger.exception("Error running ansible playbook")        raise e
def _convert_non_cli_args(self, parser_name, values_dict):    """    Casts arguments to correct types by modifying values_dict param.    By default all the values are strings.    :param parser_name: The command name, e.g. main, virsh, ospd, etc    :param values_dict: The dict of with arguments    """    for key, value in values_dict.items():        if key in self.non_cli_args:            try:                values_dict[key] = self.non_cli_args[key](value)            except (TypeError, ValueError):                raise ValueError(f"Invalid value for argument '{key}': {value!r}")    return values_dict
def get_plugin_spec_flatten_dict(plugin_dir):    """    Creates a flat dictionary from the plugin spec    :param plugin_dir: A path to the plugin's directory    :return: A flatten dictionary containing the plugin's properties    """    try:        with open(os.path.join(plugin_dir, 'plugin.spec'), 'r') as f:            spec = json.load(f)            flatten_dict = {}            for key, value in spec.items():                if isinstance(value, dict):                    flatten_dict.update(value)                else:                    flatten_dict[key] = value            return flatten_dict    except OSError as e:        raise ValueError(f"Failed to read plugin spec: {e}") from e    except json.JSONDecodeError as e:        raise ValueError(f"Failed to parse plugin spec: {e}") from e
def inject_config(self):    """Set the environment variable for config path, if it is undefined."""    # Check if the environment variable is undefined    if not os.environ.get("CONFIG_PATH"):        # Set the environment variable to the default config path        os.environ["CONFIG_PATH"] = "/path/to/default/config.ini"
def extend_cli(self, root_subparsers):    """    Adds the spec cli options to to the main entry point.    :param subparser: the subparser object to extend.    """    try:        # Get the spec options from the environment        spec_options = self.get_spec_options()        # Create a subparser for the spec options        spec_subparser = root_subparsers.add_parser('spec', help='spec command')        # Add the spec options to the subparser        for option in spec_options:            spec_subparser.add_argument(option.short, option.long, help=option.help)        # Set the default spec options        spec_subparser.set_defaults(**{option.long: option.default for option in spec_options})    except Exception as e:        # Log the error and raise it as a RuntimeError        logging.error(e)        raise RuntimeError(e)
def strip_root(path, root):    """Remove root from path, throw exception on failure.    Args:        path (str): The path to modify.        root (str): The root directory to remove from the path.    Returns:        str: The modified path with the root directory removed.    Raises:        ValueError: If the root directory is not present in the path.    """    if not path.startswith(root):        raise ValueError(f"Root directory '{root}' not present in path '{path}'.")    return path[len(root):]
