def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    from itertools import combinations    def apply_absorption(expr1, expr2):        if isinstance(expr1, tuple) and isinstance(expr2, tuple):            op1, a1, b1 = expr1            op2, a2, b2 = expr2            if op1 == '&' and op2 == '|':                if a1 == a2:                    return a1                if a1 == ('~', a2):                    return (op1, a1, b2)            if op1 == '|' and op2 == '&':                if a1 == a2:                    return a1                if a1 == ('~', a2):                    return (op1, a1, b2)        return None    new_args = args[:]    for i, j in combinations(range(len(args)), 2):        expr1, expr2 = args[i], args[j]        result = apply_absorption(expr1, expr2)        if result:            new_args.remove(expr1)            new_args.remove(expr2)            new_args.append(result)            return self.absorb(new_args)  # Recursively apply absorption    return new_args
