import java.util.Stack;public class FrameStack {    private Stack<String> outputFrameStack = new Stack<>();    /**      * Pops as many abstract types from the output frame stack as described by the given descriptor.     * @param descriptor a type or method descriptor (in which case its argument types are popped).     */    private void pop(final String descriptor) {        if (descriptor == null || descriptor.isEmpty()) {            return;        }        if (descriptor.charAt(0) == '(') {            // Method descriptor            int i = 1;            while (descriptor.charAt(i) != ')') {                i = popType(descriptor, i);            }        } else {            // Field descriptor            popType(descriptor, 0);        }    }    private int popType(final String descriptor, int index) {        switch (descriptor.charAt(index)) {            case 'B': // byte            case 'C': // char            case 'D': // double            case 'F': // float            case 'I': // int            case 'J': // long            case 'S': // short            case 'Z': // boolean                outputFrameStack.pop();                return index + 1;            case 'L': // class                int endIndex = descriptor.indexOf(';', index);                outputFrameStack.pop();                return endIndex + 1;            case '[': // array                return popType(descriptor, index + 1);            default:                throw new IllegalArgumentException("Invalid descriptor: " + descriptor);        }    }    public static void main(String[] args) {        FrameStack frameStack = new FrameStack();        // Example usage        frameStack.outputFrameStack.push("int");        frameStack.outputFrameStack.push("double");        frameStack.outputFrameStack.push("java/lang/String");        frameStack.pop("(IDLjava/lang/String;)V");        System.out.println(frameStack.outputFrameStack); // Should be empty    }}
