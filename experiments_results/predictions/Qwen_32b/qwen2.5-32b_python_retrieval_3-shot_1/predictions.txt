def cached(cache, key=hashkey, lock=None):    """Decorator to wrap a function with a memoizing callable that saves    results in a cache.    """    def decorator(func):        if cache is None:            def wrapper(*args, **kwargs):                return func(*args, **kwargs)        elif lock is None:            def wrapper(*args, **kwargs):                k = key(*args, **kwargs)                try:                    return cache[k]                except KeyError:                    pass  # key not found                v = func(*args, **kwargs)                try:                    cache[k] = v                except ValueError:                    pass  # value too large                return v        else:            def wrapper(*args, **kwargs):                k = key(*args, **kwargs)                try:                    with lock:                        return cache[k]                except KeyError:                    pass  # key not found                v = func(*args, **kwargs)                try:                    with lock:                        cache[k] = v                except ValueError:                    pass  # value too large                return v        return _update_wrapper(wrapper, func)    return decorator

"""    def decorator(func):        cache = lru_cache(maxsize=maxsize, typed=typed)        timestamps = {}        @wraps(func)        def wrapper(*args, **kwargs):            key = (args, tuple(sorted(kwargs.items()))) if typed else args + tuple(kwargs.values())            current_time = timer()            if key in timestamps and current_time - timestamps[key] > ttl:                cache.cache_clear()                timestamps.pop(key, None)            if key not in timestamps:                result = func(*args, **kwargs)                cache.cache_clear()                cache(key)                timestamps[key] = current_time                return result            return cache(key)        return wrapper    return decorator
from functools import wrapsfrom collections import OrderedDictdef mru_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Most Recently Used (MRU)    algorithm.    """    def decorator(func):        cache = OrderedDict()        cache_get = cache.get        cache_len = cache.__len__        cache_popitem = cache.popitem        cache_setitem = cache.__setitem__        @wraps(func)        def wrapper(*args, **kwargs):            if kwargs:                key = args + (typed and type(args),) + tuple(sorted(kwargs.items()))            else:                key = args + (typed and type(args),)            if key in cache:                cache.move_to_end(key)                return cache_get(key)            result = func(*args, **kwargs)            if cache_len() >= maxsize:                cache_popitem(last=False)            cache_setitem(key, result)            return result        def cache_clear():            cache.clear()        wrapper.cache_clear = cache_clear        return wrapper    return decorator

from collections import defaultdictfrom functools import wrapsdef lfu_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Least Frequently Used (LFU)    algorithm.    """    def decorator(func):        cache = {}        frequency = defaultdict(int)        cache_get = cache.get        cache_len = cache.__len__        @wraps(func)        def wrapper(*args, **kwargs):            if typed:                key = (args, tuple(sorted(kwargs.items())))            else:                key = args + tuple(sorted(kwargs.items()))            if key in cache:                frequency[key] += 1                return cache[key]            result = func(*args, **kwargs)            if cache_len() >= maxsize:                lfu_key = min(cache, key=frequency.get)                del cache[lfu_key]                del frequency[lfu_key]            cache[key] = result            frequency[key] = 1            return result        return wrapper    return decorator

def _legacy_mergeOrderings(orderings):    """Merge multiple orderings so that within-ordering order is preserved    Orderings are constrained in such a way that if an object appears    in two or more orderings, then the suffix that begins with the    object must be in both orderings.    For example:    >>> _mergeOrderings([    ... ['x', 'y', 'z'],    ... ['q', 'z'],    ... [1, 3, 5],    ... ['z']    ... ])    ['x', 'y', 'q', 1, 3, 5, 'z']    """    from collections import OrderedDict    merged_ordering = []    orderings = [list(ordering) for ordering in orderings]  # Convert to lists if not already    while any(orderings):        # Find the first element that appears only once across all orderings        first_elements = [ordering[0] for ordering in orderings if ordering]        for element in first_elements:            if first_elements.count(element) == 1:                # Add this unique element to the merged ordering                merged_ordering.append(element)                # Remove this element from all orderings                for ordering in orderings:                    if ordering and ordering[0] == element:                        ordering.pop(0)                break        else:            # If no unique element is found, it means we have a cycle or all elements are the same            # In this case, we can just take the first element from the first non-empty ordering            for ordering in orderings:                if ordering:                    merged_ordering.append(ordering.pop(0))                    break    return merged_ordering

def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    from sympy import And, Or, Not, simplify    def apply_absorption(expr):        if isinstance(expr, And):            a, b = expr.args            if isinstance(b, Or) and a in b.args:                return a            if isinstance(b, Or) and Not(a) in b.args:                return And(a, *[arg for arg in b.args if arg != Not(a)])        elif isinstance(expr, Or):            a, b = expr.args            if isinstance(b, And) and a in b.args:                return a            if isinstance(b, And) and Not(a) in b.args:                return Or(a, *[arg for arg in b.args if arg != Not(a)])        return expr    return [simplify(apply_absorption(arg)) for arg in args]

def int_to_string(number: int, alphabet: List[str], padding: Optional[int] = None) -> str:    """    Convert a number to a string, using the given alphabet.    The output has the most significant digit first.    """    base = len(alphabet)    if number == 0:        return alphabet[0] * (padding if padding is not None else 1)        digits = []    while number:        digits.append(alphabet[number % base])        number //= base        if padding is not None:        digits += [alphabet[0]] * (padding - len(digits))        return ''.join(reversed(digits))

def verify_relayable_signature(public_key, doc, signature):    """    Verify the signed XML elements to have confidence that the claimed    author did actually generate this message.    """    from xml.etree import ElementTree    from xml.dom import minidom    from Crypto.Signature import pkcs1_15    from Crypto.Hash import SHA256    from Crypto.PublicKey import RSA    # Parse the XML document    root = ElementTree.fromstring(doc)    # Convert the ElementTree to a string    doc_str = minidom.parseString(ElementTree.tostring(root)).toprettyxml()    # Hash the document    h = SHA256.new(doc_str.encode('utf-8'))    # Import the public key    key = RSA.import_key(public_key)    # Verify the signature    try:        pkcs1_15.new(key).verify(h, signature)        return True    except (ValueError, TypeError):        return False
def parse_diaspora_webfinger(document: str) -> Dict:    """    Parse Diaspora webfinger which is either in JSON format (new) or XRD (old).    https://diaspora.github.io/diaspora_federation/discovery/webfinger.html    """    try:        data = json.loads(document)        return {            "subject": data["subject"],            "aliases": data.get("aliases", []),            "links": {link["rel"]: link["href"] for link in data.get("links", [])}        }    except json.JSONDecodeError:        try:            root = ET.fromstring(document)            subject = root.find("{http://docs.oasis-open.org/ns/xri/xrd-1.0}Subject").text            aliases = [alias.text for alias in root.findall("{http://docs.oasis-open.org/ns/xri/xrd-1.0}Alias")]            links = {}            for link in root.findall("{http://docs.oasis-open.org/ns/xri/xrd-1.0}Link"):                rel = link.get("rel")                href = link.get("href")                if rel and href:                    links[rel] = href            return {                "subject": subject,                "aliases": aliases,                "links": links            }        except ET.ParseError:            return {}
def try_retrieve_webfinger_document(handle: str) -> Optional[str]:    """    Try to retrieve an RFC7033 webfinger document. Does not raise if it fails.    """    try:        host = handle.split("@")[1]        url = f"https://{host}/.well-known/webfinger?resource=acct:{quote(handle)}"        with closing(urllib.request.urlopen(url)) as response:            return response.read().decode('utf-8')    except (IndexError, urllib.error.URLError):        return None

def ansible_playbook(ir_workspace, ir_plugin, playbook_path, verbose=None,                     extra_vars=None, ansible_args=None):    """Wraps the 'ansible-playbook' CLI.     :param ir_workspace: An Infrared Workspace object represents the active     workspace     :param ir_plugin: An InfraredPlugin object of the current plugin     :param playbook_path: the playbook to invoke     :param verbose: Ansible verbosity level     :param extra_vars: dict. Passed to Ansible as extra-vars     :param ansible_args: dict of ansible-playbook arguments to plumb down         directly to Ansible.    """    command = ["ansible-playbook"]        if verbose:        command.append(f"-{'v' * verbose}")        if extra_vars:        extra_vars_str = " ".join([f"{k}={v}" for k, v in extra_vars.items()])        command.extend(["--extra-vars", extra_vars_str])        if ansible_args:        for arg, value in ansible_args.items():            if value is not None:                command.extend([f"--{arg}", str(value)])            else:                command.append(f"--{arg}")        command.append(playbook_path)        logger.debug("Executing Ansible call `%s`", " ".join(command))    p = subprocess.Popen(command)    p.communicate()    if p.returncode != 0:        raise RuntimeError(f"Ansible reported an error when executing playbook {playbook_path}")
