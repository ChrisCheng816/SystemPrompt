from collections import OrderedDictfrom functools import wrapsdef mru_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Most Recently Used (MRU)    algorithm.    """    def decorator(func):        cache = OrderedDict()        cache_hits = 0        cache_misses = 0        @wraps(func)        def wrapper(*args, **kwargs):            nonlocal cache_hits, cache_misses            if typed:                key = (args, tuple(sorted(kwargs.items())))            else:                key = (args, frozenset(kwargs.items()))            if key in cache:                cache_hits += 1                # Move the key to the end to show that it was recently used                cache.move_to_end(key)                return cache[key]            else:                cache_misses += 1                result = func(*args, **kwargs)                cache[key] = result                if len(cache) > maxsize:                    # Pop the last item (most recently used)                    cache.popitem(last=False)                return result        def cache_info():            """Report cache statistics"""            return (cache_hits, cache_misses, maxsize, len(cache))        def cache_clear():            """Clear the cache and cache statistics"""            nonlocal cache_hits, cache_misses            cache.clear()            cache_hits = 0            cache_misses = 0        wrapper.cache_info = cache_info        wrapper.cache_clear = cache_clear        return wrapper    return decorator

def _legacy_mergeOrderings(orderings):    """Merge multiple orderings so that within-ordering order is preserved    Orderings are constrained in such a way that if an object appears    in two or more orderings, then the suffix that begins with the    object must be in both orderings. """    command = ["ansible-playbook", playbook_path]    if verbose:        command.append(f"-{'v' * verbose}")    if extra_vars:        extra_vars_str = " ".join([f"{key}={value}" for key, value in extra_vars.items()])        command.extend(["--extra-vars", extra_vars_str])    if ansible_args:        for arg, value in ansible_args.items():            if value is None:                command.append(f"--{arg}")            else:                command.extend([f"--{arg}", str(value)])    logger.debug("Executing Ansible command: %s", " ".join(command))    process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.PIPE)    stdout, stderr = process.communicate()    if process.returncode != 0:        raise RuntimeError(f"Ansible playbook failed with return code {process.returncode}: {stderr.decode()}")    return stdout.decode()
def _run_playbook(self, cli_args, vars_dict, ir_workspace, ir_plugin):    """Runs ansible cli with vars dict    :param vars_dict: dict, Will be passed as Ansible extra-vars    :param cli_args: the list of command line arguments    :param ir_workspace: An Infrared Workspace object represents the active     workspace    :param ir_plugin: An InfraredPlugin object of the current plugin    :return: ansible results    """    # Construct the command with extra-vars    extra_vars = " ".join([f"{key}={value}" for key, value in vars_dict.items()])    command = ["ansible-playbook"] + cli_args + ["--extra-vars", extra_vars]    # Log the command being executed    logger.debug("Executing Ansible command: %s", " ".join(command))    # Execute the command    process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.PIPE)    stdout, stderr = process.communicate()    # Check for errors    if process.returncode != 0:        logger.error("Ansible command failed with error: %s", stderr.decode())        raise RuntimeError(f"Ansible command failed with error: {stderr.decode()}")    # Return the results    return stdout.decode()

def get_plugin_spec_flatten_dict(plugin_dir):    """Creates a flat dict from the plugin spec    :param plugin_dir: A path to the plugin's dir    :return: A flatten dictionary contains the plugin's properties    """    import os    import json    # Assuming the plugin spec is in a file named 'plugin_spec.json' within the plugin_dir    spec_file_path = os.path.join(plugin_dir, 'plugin_spec.json')        if not os.path.exists(spec_file_path):        raise FileNotFoundError(f"Plugin spec file not found at {spec_file_path}")    with open(spec_file_path, 'r') as file:        plugin_spec = json.load(file)    def flatten_dict(d, parent_key='', sep='_'):        items = []        for k, v in d.items():            new_key = f"{parent_key}{sep}{k}" if parent_key else k            if isinstance(v, dict):                items.extend(flatten_dict(v, new_key, sep=sep).items())            else:                items.append((new_key, v))        return dict(items)    return flatten_dict(plugin_spec)

def strip_root(path, root):    """    Remove root from path, throw exception on failure.    :param path: The path from which to remove the root.    :param root: The root to remove from the path.    :return: The path with the root removed.    :raises ValueError: If the root is not a prefix of the path.
