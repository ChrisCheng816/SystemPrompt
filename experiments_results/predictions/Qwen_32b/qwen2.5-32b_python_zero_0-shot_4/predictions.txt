Args:            hook (str): Hook attribute for which to register the handler.        Returns:            callable: Decorator for registering listeners for the specified hook.        """        def decorator(func: Callable) -> Callable:            if hook not in self._registry:                self._registry[hook] = []            self._registry[hook].append(func)            return func        return decorator    def trigger(self, hook: str, *args: Any, **kwargs: Any) -> None:        """Trigger all handlers registered for a specific hook. """        if hook in self._registry:            for func in self._registry[hook]:                func(*args, **kwargs)# Example usageif __name__ == "__main__":    registry = HookRegistry()    @registry.on("start")    def handle_start():        print("Starting...")    @registry.on("start")    def handle_start_with_message(message: str):        print(f"Starting with message: {message}")    @registry.on("end")    def handle_end():        print("Ending...")    registry.trigger("start")    registry.trigger("start", message="Hello, World!")    registry.trigger("end")
