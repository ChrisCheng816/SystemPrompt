import java.nio.ByteBuffer;import java.nio.charset.StandardCharsets;private static int decodeOctets(int i, ByteBuffer bb, StringBuilder sb) {    while (bb.hasRemaining()) {        int b = bb.get() & 0xFF;        if ((b & 0x80) == 0) {            // Single-byte character            sb.append((char) b);            i++;        } else if ((b & 0xE0) == 0xC0) {            // Two-byte character            if (bb.hasRemaining()) {                int b2 = bb.get() & 0xFF;                sb.append((char) ((b & 0x1F) << 6 | (b2 & 0x3F)));                i += 2;            } else {                break;            }        } else if ((b & 0xF0) == 0xE0) {            // Three-byte character            if (bb.remaining() >= 2) {                int b2 = bb.get() & 0xFF;                int b3 = bb.get() & 0xFF;                sb.append((char) ((b & 0x0F) << 12 | (b2 & 0x3F) << 6 | (b3 & 0x3F)));                i += 3;            } else {                break;            }        } else if ((b & 0xF8) == 0xF0) {            // Four-byte character            if (bb.remaining() >= 3) {                int b2 = bb.get() & 0xFF;                int b3 = bb.get() & 0xFF;                int b4 = bb.get() & 0xFF;                int codePoint = ((b & 0x07) << 18 | (b2 & 0x3F) << 12 | (b3 & 0x3F) << 6 | (b4 & 0x3F));                sb.append(Character.toChars(codePoint));                i += 4;            } else {                break;            }        } else {            // Invalid UTF-8 sequence            break;        }    }    return i;}
