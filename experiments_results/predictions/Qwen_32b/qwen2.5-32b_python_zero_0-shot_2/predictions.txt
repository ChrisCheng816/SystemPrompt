"""        if not os.path.exists(path):            return False        try:            with open(os.path.join(path, 'ocfl_object_1.0'), 'r') as f:                if f.read().strip() != 'ocfl_object_1.0':                    return False        except FileNotFoundError:            return False        try:            with open(os.path.join(path, 'inventory.json'), 'r') as f:                inventory = json.load(f)        except (FileNotFoundError, json.JSONDecodeError):            return False        if not self._validate_inventory(inventory):            return False        for digest_algorithm in inventory.get('digestAlgorithm', []):            if not self._validate_files(path, inventory, digest_algorithm):                return False        return True    def _validate_inventory(self, inventory):        required_keys = {'id', 'type', 'digestAlgorithm', 'head', 'versions'}        if not required_keys.issubset(inventory.keys()):            return False        if inventory['type'] != 'https://ocfl.io/1.0/spec/#inventory':            return False        return True    def _validate_files(self, path, inventory, digest_algorithm):        for version in inventory['versions'].values():            for file_path, file_info in version['state'].items():                file_digest = file_info.get(digest_algorithm)                if not file_digest:                    return False                full_path = os.path.join(path, file_path)                if not os.path.exists(full_path):                    return False                if not self._check_file_digest(full_path, digest_algorithm, file_digest):                    return False        return True    def _check_file_digest(self, file_path, digest_algorithm, expected_digest):        hash_func = hashlib.new(digest_algorithm)        with open(file_path, 'rb') as f:            for chunk in iter(lambda: f.read(4096), b""):                hash_func.update(chunk)        return hash_func.hexdigest() == expected_digest
