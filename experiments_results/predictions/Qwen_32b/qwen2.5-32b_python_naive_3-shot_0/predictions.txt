def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    new_args = []    for expr in args:        if isinstance(expr, tuple) and len(expr) == 3:            op, left, right = expr            if op == '&':                if isinstance(right, tuple) and right[0] == '|':                    if left == right[1] or left == right[2]:                        new_args.append(left)                        continue                    elif isinstance(right[1], tuple) and right[1][0] == '~' and right[1][1] == left:                        new_args.append(('&', left, right[2]))                        continue                    elif isinstance(right[2], tuple) and right[2][0] == '~' and right[2][1] == left:                        new_args.append(('&', left, right[1]))                        continue            elif op == '|':                if isinstance(right, tuple) and right[0] == '&':                    if left == right[1] or left == right[2]:                        new_args.append(left)                        continue                    elif isinstance(right[1], tuple) and right[1][0] == '~' and right[1][1] == left:                        new_args.append(('|', left, right[2]))                        continue                    elif isinstance(right[2], tuple) and right[2][0] == '~' and right[2][1] == left:                        new_args.append(('|', left, right[1]))                        continue        new_args.append(expr)    return new_args
