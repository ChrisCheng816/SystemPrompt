def difference(d1, d2, level=-1):    """Return a dictionary with items from *d1* not contained in *d2*.    *level* sets the maximum depth of recursion. For infinite recursion,    set that to -1. See :func:`intersection` for more details.    *d1* and *d2* remain unchanged. However, *d1* or some of its    subdictionaries may be returned directly. """    if level == 0:        return {}    diff = {}    for key in d1:        if key not in d2:            diff[key] = d1[key]        elif isinstance(d1[key], dict) and isinstance(d2[key], dict):            sub_diff = difference(d1[key], d2[key], level - 1)            if sub_diff:                diff[key] = sub_diff        elif d1[key] != d2[key]:            diff[key] = d1[key]    return diff

class ExpressionAbsorber:    def __init__(self):        pass    def absorb(self, args):        """        Given an `args` sequence of expressions, return a new list of expression        applying absorption and negative absorption.        See https://en.wikipedia.org/wiki/Absorption_law        Absorption::            A & (A | B) = A, A | (A & B) = A        Negative absorption::            A & (~A | B) = A & B, A | (~A & B) = A | B        """        from collections import deque        def apply_absorption(expr):            stack = deque()            operators = {'&', '|'}            for term in expr:                if term in operators:                    stack.append(term)                else:                    if len(stack) >= 3 and stack[-1] in operators:                        op1 = stack.pop()                        b = stack.pop()                        op2 = stack.pop()                        a = stack.pop()                        if op1 == '|' and op2 == '&' and a == b:                            stack.append(a)                        elif op1 == '&' and op2 == '|' and a == b:                            stack.append(a)                        elif op1 == '|' and op2 == '&' and a == f'~{b}':                            stack.append(f'{a} & {b}')                        elif op1 == '&' and op2 == '|' and a == f'~{b}':                            stack.append(f'{a} | {b}')                        else:                            stack.append(a)                            stack.append(op2)                            stack.append(b)                            stack.append(op1)                    stack.append(term)            return list(stack)        return [apply_absorption(expr) for expr in args]# Example usage:# absorber = ExpressionAbsorber()# print(absorber.absorb([['A', '&', '(', 'A', '|', 'B', ')']]))# print(absorber.absorb([['A', '|', '(', 'A', '&', 'B', ')']]))# print(absorber.absorb([['A', '&', '(', '~A', '|', 'B', ')']]))# print(absorber.absorb([['A', '|', '(', '~A', '&', 'B', ')']]))
