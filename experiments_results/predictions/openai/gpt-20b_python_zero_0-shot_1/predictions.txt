def size_to_bytes(size: str) -> int:    """Convert human readable file size to bytes.    Resulting value is an approximation as input value is in most case rounded.    Args:        size: A string representing a human readable file size (eg: '500K')    Returns:        A decimal representation of file size        Examples::            >>> size_to_bytes("500")            500            >>> size_to_bytes("1K")            1000    """    import re    size = size.strip()    if not size:        raise ValueError("size string is empty")    # Regex to capture numeric part and optional unit    match = re.fullmatch(r'(?P<num>[0-9]*\.?[0-9]+)\s*(?P<unit>[kKmMgGtTpPeE]?)', size)    if not match:        raise ValueError(f"Invalid size format: {size!r}")    num = float(match.group('num'))    unit = match.group('unit').lower()    multipliers = {        '': 1,        'k': 10**3,        'm': 10**6,        'g': 10**9,        't': 10**12,        'p': 10**15,        'e': 10**18,    }    if unit not in multipliers:        raise ValueError(f"Unknown unit: {unit!r}")    return int(round(num * multipliers[unit]))

@staticmethod    def merge_extra_vars(vars_dict, extra_vars=None):        """        Extend ``vars_dict`` with ``extra-vars``.        :param vars_dict: Dictionary to merge extra-vars into        :param extra_vars: List of extra-vars strings (e.g., ["key=value", "list=1,2,3"])        :return: The updated dictionary        """        if not extra_vars:            return vars_dict        for item in extra_vars:            if not isinstance(item, str):                continue            if "=" not in item:                continue            key, value = item.split("=", 1)            key = key.strip()            value = value.strip()            if not key:                continue            # If value contains commas, split into list            if "," in value:                value = [v.strip() for v in value.split(",") if v.strip()]            vars_dict[key] = value        return vars_dict
