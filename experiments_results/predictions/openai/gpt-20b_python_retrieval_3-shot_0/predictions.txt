def _dictsum(dicts):    """    Combine values of the dictionaries supplied by iterable ``dicts``.    The function sums the values of each key across all dictionaries. Returns    -------    dict        A dictionary containing each key that appears in any of the input        dictionaries, with the values summed across all dictionaries.    Examples    --------    >>> _dictsum([{'a': 1, 'b': 2}, {'a': 5, 'b': 0}])    {'a': 6, 'b': 2}    """    from collections import Counter    from functools import reduce    import operator    # Convert each dictionary to a Counter so that missing keys are treated    # as zero and numeric addition works naturally.    counters = [Counter(d) for d in dicts]    # If no dictionaries were supplied, return an empty dict.    if not counters:        return {}    # Reduce the list of counters by adding them together.    summed = reduce(operator.add, counters)    # Convert back to a plain dict before returning.    return dict(summed)
