def pretty(self, indent=0, debug=False):    """    Return a pretty formatted representation of self. def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    def apply_absorption(expr):        if isinstance(expr, And):            if expr.left == expr.right.left:                return expr.left            elif expr.left == expr.right.right:                return expr.left        elif isinstance(expr, Or):            if expr.left == expr.right.left:                return expr.left            elif expr.left == expr.right.right:                return expr.left        return expr    def apply_negative_absorption(expr):        if isinstance(expr, And):            if isinstance(expr.right, Or) and expr.left == expr.right.left:                return And(expr.left, expr.right.right)            elif isinstance(expr.right, Or) and expr.left == expr.right.right:                return And(expr.left, expr.right.left)        elif isinstance(expr, Or):            if isinstance(expr.right, And) and expr.left == expr.right.left:                return Or(expr.left, expr.right.right)            elif isinstance(expr.right, And) and expr.left == expr.right.right:                return Or(expr.left, expr.right.left)        return expr    new_args = []    for arg in args:        arg = apply_absorption(arg)        arg = apply_negative_absorption(arg)        new_args.append(arg)    return new_args

The output has the most significant digit first.    """    if number < 0:        raise ValueError("Number must be non-negative")        base = len(alphabet)    if base < 2:        raise ValueError("Alphabet must contain at least two characters")        if number == 0:        return alphabet[0]        digits = []    while number:        digits.append(alphabet[number % base])        number //= base        if padding is not None:        digits.extend([alphabet[0]] * (padding - len(digits)))        return ''.join(reversed(digits))

By default all the values are strings.    :param parser_name: The command name, e.g. main, virsh, ospd, etc    :param values_dict: The dict of with arguments    """    # Retrieve the parser based on the parser_name    parser = self.parsers.get(parser_name)    if not parser:        raise ValueError(f"No parser found for the name: {parser_name}")    # Iterate over each argument in the parser    for action in parser._actions:        if action.dest in values_dict:            # Convert the value to the correct type            if action.type:                values_dict[action.dest] = action.type(values_dict[action.dest])            elif action.choices:                # If choices are provided, ensure the value is one of them                if values_dict[action.dest] not in action.choices:                    raise ValueError(f"Invalid value for {action.dest}: {values_dict[action.dest]}")            elif action.nargs == '?':                # If nargs is '?', the value can be None                if values_dict[action.dest] is not None and action.type:                    values_dict[action.dest] = action.type(values_dict[action.dest])            elif action.nargs == '+':                # If nargs is '+', ensure all values are of the correct type                if action.type:                    values_dict[action.dest] = [action.type(v) for v in values_dict[action.dest]]            elif action.nargs == '*':                # If nargs is '*', ensure all values are of the correct type                if action.type:                    values_dict[action.dest] = [action.type(v) for v in values_dict[action.dest]]

def strip_root(path, root):    """Remove root from path, throw exception on failure."""    if not path.startswith(root):        raise ValueError("Path does not start with the specified root.")        stripped_path = path[len(root):]    if stripped_path.startswith(os.sep):        stripped_path = stripped_path[1:]        return stripped_path
