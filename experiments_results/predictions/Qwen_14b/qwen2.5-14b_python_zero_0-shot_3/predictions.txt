"""    if level == 0:        return d1.copy()        diff = {}    for key in d1:        if key not in d2:            diff[key] = d1[key]        elif isinstance(d1[key], dict) and isinstance(d2[key], dict):            sub_diff = difference(d1[key], d2[key], level - 1)            if sub_diff:                diff[key] = sub_diff        elif d1[key] != d2[key]:            diff[key] = d1[key]        return diff
