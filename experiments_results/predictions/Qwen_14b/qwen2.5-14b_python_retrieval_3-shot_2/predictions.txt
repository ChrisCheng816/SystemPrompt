def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    result = []    for expr in args:        if isinstance(expr, And):            if expr.left == expr.right.left:                result.append(expr.left)            elif expr.left == expr.right.right:                result.append(expr.left)            elif isinstance(expr.right, Or) and expr.left == ~expr.right.left:                result.append(And(expr.left, expr.right.right))            elif isinstance(expr.right, Or) and expr.left == ~expr.right.right:                result.append(And(expr.left, expr.right.left))        elif isinstance(expr, Or):            if expr.left == expr.right.left:                result.append(expr.left)            elif expr.left == expr.right.right:                result.append(expr.left)            elif isinstance(expr.right, And) and expr.left == ~expr.right.left:                result.append(Or(expr.left, expr.right.right))            elif isinstance(expr.right, And) and expr.left == ~expr.right.right:                result.append(Or(expr.left, expr.right.left))        else:            result.append(expr)    return result
