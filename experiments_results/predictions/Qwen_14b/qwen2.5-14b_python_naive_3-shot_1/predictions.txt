def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    result = []    for expr in args:        if isinstance(expr, And):            if expr.left == expr.right.left or expr.left == expr.right.right:                result.append(expr.left)            elif expr.right.left == Not(expr.left) or expr.right.right == Not(expr.left):                result.append(And(expr.left, expr.right.right if expr.right.left == Not(expr.left) else expr.right.left))        elif isinstance(expr, Or):            if expr.left == expr.right.left or expr.left == expr.right.right:                result.append(expr.left)            elif expr.right.left == Not(expr.left) or expr.right.right == Not(expr.left):                result.append(Or(expr.left, expr.right.right if expr.right.left == Not(expr.left) else expr.right.left))        else:            result.append(expr)    return result

Tries to ignore tags inside code blocks.    Optionally, if passed a "replacer", will also replace the tag word with the result    of the replacer function called with the tag word.
