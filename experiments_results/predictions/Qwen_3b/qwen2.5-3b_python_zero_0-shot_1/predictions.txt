from collections import defaultdictdef lfu_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Least Frequently Used (LFU)    algorithm.    """    cache = defaultdict(lambda: (0, None))    usage = defaultdict(int)    def decorator(func):        def wrapper(*args, **kwargs):            key = (args, frozenset(kwargs.items())) if typed else args            if key in cache:                usage[key] += 1                return cache[key][1]            result = func(*args, **kwargs)            cache[key] = (usage[key], result)            usage[key] += 1            if len(cache) > maxsize:                min_usage = min(usage.values())                for k, v in cache.items():                    if v[0] == min_usage:                        del cache[k]                        del usage[k]                        break            return result        return wrapper    return decorator
