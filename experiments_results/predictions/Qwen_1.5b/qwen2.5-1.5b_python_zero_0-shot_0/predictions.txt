"""    tags = set()    in_code_block = False    current_tag = None    for char in text:        if char == '<':            in_code_block = True            current_tag = None        elif char == '>':            in_code_block = False            if current_tag:                tags.add(current_tag)                if replacer:                    current_tag = replacer(current_tag)        elif not in_code_block and char.isalpha():            if current_tag is None:                current_tag = char            else:                current_tag += char    return tags, text
