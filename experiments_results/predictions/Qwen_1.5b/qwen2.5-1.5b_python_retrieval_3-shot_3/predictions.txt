def subclasses(cls):    """Return all subclasses of a class, recursively"""    seen = set()    queue = set([cls])    while queue:        c = queue.pop()        seen.add(c)        sc = c.__subclasses__()        for c in sc:            if c not in seen:                queue.add(c)    seen.remove(cls)    return seen

The input is assumed to have the most significant digit first.    """    base = len(alphabet)    inverse_alphabet = dict(zip(alphabet, xrange(0, base)))    n = 0    exp = 0    for i in reversed(string):        n += inverse_alphabet[i] * (base ** exp)        exp += 1    return n
