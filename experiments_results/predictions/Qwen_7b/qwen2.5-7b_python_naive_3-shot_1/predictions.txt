It maps each parameter    (:class:`str` or :class:`int`) to value. If :attr:`.SQLParams.in_style`    is a named parameter style. then *params* must be a :class:`~collections.abc.Mapping`.    If :attr:`.SQLParams.in_style` is an ordinal parameter style, then    *params* must be a :class:`~collections.abc.Sequence`. """    if self.SQLParams.in_style == 'named':        placeholders = {f':{k}': v for k, v in params.items()}    elif self.SQLParams.in_style == 'ordinal':        placeholders = {f'${i+1}': v for i, v in enumerate(params)}    else:        raise ValueError("Unsupported parameter style")    formatted_sql = sql    for placeholder, value in placeholders.items():        formatted_sql = formatted_sql.replace(placeholder, str(value))    return formatted_sql, placeholders

"""    if isinstance(sql, bytes):        sql = sql.decode('utf-8')    out_params = []    for params in many_params:        if isinstance(params, Mapping):            out_params.append({f'out_{k}': v for k, v in params.items()})        elif isinstance(params, Sequence):            out_params.append([f'out_{i}' for i in range(len(params))])        else:            raise ValueError("Invalid parameter type")    formatted_sql = sql.format(**{f'out_{i}': f'{{{i}}}' for i in range(len(out_params[0]))})    return formatted_sql, out_params

def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    new_args = []    for arg in args:        if isinstance(arg, tuple) and len(arg) == 2:            op, expr = arg            if op == '&':                if isinstance(expr, tuple) and expr[0] == '|':                    if expr[1] == arg[0]:                        new_args.append(arg[0])                    elif expr[0] == arg[0]:                        new_args.append(expr[1])                elif isinstance(expr, tuple) and expr[0] == '&':                    if expr[1] == arg[0]:                        new_args.append(expr[1])                    elif expr[0] == arg[0]:                        new_args.append(expr[1])            elif op == '|':                if isinstance(expr, tuple) and expr[0] == '&':                    if expr[1] == arg[0]:                        new_args.append(arg[0])                    elif expr[0] == arg[0]:                        new_args.append(expr[1])                elif isinstance(expr, tuple) and expr[0] == '|':                    if expr[1] == arg[0]:                        new_args.append(expr[1])                    elif expr[0] == arg[0]:                        new_args.append(expr[1])    return new_args
