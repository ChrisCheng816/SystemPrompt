from functools import wraps, partialfrom collections import OrderedDictdef mru_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Most Recently Used (MRU)    algorithm.    """    def decorator(func):        cache = OrderedDict()        hit = 0        miss = 0        @wraps(func)        def wrapper(*args, **kwargs):            nonlocal hit, miss            key = (args, tuple(kwargs.items())) if typed else args            if key in cache:                hit += 1                cache.move_to_end(key)                return cache[key]            else:                miss += 1                result = func(*args, **kwargs)                cache[key] = result                if len(cache) > maxsize:                    cache.popitem(last=False)                return result        wrapper.cache_info = lambda: (hit, miss, maxsize, len(cache))        return wrapper    return decorator
def lru_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Least Recently Used (LRU)    algorithm.    """    def decorator(func):        cache = {}        cache_order = []        def wrapper(*args, **kwargs):            key = (args, tuple(kwargs.items())) if typed else args            if key in cache:                cache_order.remove(key)                cache_order.append(key)                return cache[key]            elif len(cache) >= maxsize:                oldest_key = cache_order.pop(0)                del cache[oldest_key]            result = func(*args, **kwargs)            cache[key] = result            cache_order.append(key)            return result        return wrapper    return decorator
def lfu_cache(maxsize=128, typed=False):    """Decorator to wrap a function with a memoizing callable that saves    up to `maxsize` results based on a Least Frequently Used (LFU)    algorithm.    """    from collections import OrderedDict, defaultdict    class Node:        def __init__(self, key, value, frequency=1):            self.key = key            self.value = value            self.frequency = frequency    class LFUCache:        def __init__(self, maxsize, typed):            self.maxsize = maxsize            self.typed = typed            self.cache = OrderedDict()            self.frequency = defaultdict(OrderedDict)            self.min_freq = 0        def __getitem__(self, key):            node = self.cache[key]            self._update_frequency(node)            return node.value        def __setitem__(self, key, value):            if key in self.cache:                node = self.cache[key]                node.value = value                self._update_frequency(node)            else:                if len(self.cache) >= self.maxsize:                    self._evict()                node = Node(key, value)                self.cache[key] = node                self.frequency[node.frequency][key] = node                self.min_freq = 1        def _update_frequency(self, node):            freq = node.frequency            del self.frequency[freq][node.key]            if not self.frequency[freq]:                del self.frequency[freq]                if freq == self.min_freq:                    self.min_freq += 1            node.frequency += 1            self.frequency[node.frequency][node.key] = node        def _evict(self):            key, node = self.frequency[self.min_freq].popitem(last=False)            del self.cache[key]    def decorator(func):        cache = LFUCache(maxsize, typed)        def wrapper(*args, **kwargs):            key = (func, args, tuple(kwargs.items())) if typed else (func, args, frozenset(kwargs.items()))            try:                return cache[key]            except KeyError:                value = func(*args, **kwargs)                cache[key] = value                return value        return wrapper    return decorator

def difference(d1, d2, level=-1):    """Return a dictionary with items from *d1* not contained in *d2*.    *level* sets the maximum depth of recursion. For infinite recursion,    set that to -1. """    if level == 0:        return {}    if not isinstance(d1, dict) or not isinstance(d2, dict):        return d1    result = {}    for key in d1:        if key not in d2:            result[key] = d1[key]        elif isinstance(d1[key], dict) and isinstance(d2[key], dict):            diff = difference(d1[key], d2[key], level - 1)            if diff:                result[key] = diff        elif d1[key] != d2[key]:            result[key] = d1[key]    return result

def absorb(self, args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    def apply_absorption(expr):        if isinstance(expr, tuple) and len(expr) == 3 and expr[0] in ('&', '|'):            op, a, b = expr            if op == '&':                if a == b:                    return a                if isinstance(b, tuple) and b[0] == '|':                    if b[1] == a or b[2] == a:                        return a            elif op == '|':                if a == b:                    return a                if isinstance(b, tuple) and b[0] == '&':                    if b[1] == a or b[2] == a:                        return a        return expr    def apply_negative_absorption(expr):        if isinstance(expr, tuple) and len(expr) == 3 and expr[0] in ('&', '|'):            op, a, b = expr            if op == '&':                if isinstance(b, tuple) and b[0] == '|':                    if b[1] == a:                        return b[2]                    if b[2] == a:                        return b[1]            elif op == '|':                if isinstance(b, tuple) and b[0] == '&':                    if b[1] == a:                        return b[2]                    if b[2] == a:                        return b[1]        return expr    new_args = []    for arg in args:        new_arg = apply_absorption(arg)        new_arg = apply_negative_absorption(new_arg)        new_args.append(new_arg)    return new_args

"""    from xml.etree import ElementTree as ET    from Crypto.PublicKey import RSA    from Crypto.Signature import PKCS1_v1_5    from Crypto.Hash import SHA256    # Parse the XML document    root = ET.fromstring(doc)    # Extract the signed element    signed_element = root.find('.//{http://www.w3.org/2000/09/xmldsig#}Signature')    # Extract the signature value    signature_value = base64.b64decode(signed_element.find('.//{http://www.w3.org/2000/09/xmldsig#}SignatureValue').text)    # Extract the canonicalized signed element    canonicalized_signed_element = signed_element.find('.//{http://www.w3.org/2000/09/xmldsig#}SignedInfo').text    # Create a hash object    hash_obj = SHA256.new(canonicalized_signed_element.encode('utf-8'))    # Load the public key    key = RSA.import_key(public_key)    # Create a verifier object    verifier = PKCS1_v1_5.new(key)    # Verify the signature    return verifier.verify(hash_obj, signature_value)
