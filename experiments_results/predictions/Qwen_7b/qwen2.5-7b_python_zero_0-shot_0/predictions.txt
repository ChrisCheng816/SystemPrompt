"""    class LFUCache:        def __init__(self, maxsize, typed):            self.maxsize = maxsize            self.typed = typed            self.cache = OrderedDict()            self.frequency = {}            self.size = 0        def __call__(self, func):            @wraps(func)            def wrapper(*args, **kwargs):                key = (func, args, tuple(kwargs.items())) if self.typed else (func, tuple(args), tuple(kwargs.items()))                if key in self.cache:                    self.frequency[key] += 1                    self.cache.move_to_end(key)                else:                    if self.size >= self.maxsize:                        lfu_key = min(self.frequency, key=self.frequency.get)                        del self.cache[lfu_key]                        del self.frequency[lfu_key]                        self.size -= 1                    self.cache[key] = func(*args, **kwargs)                    self.frequency[key] = 1                    self.size += 1                return self.cache[key]            return wrapper    return LFUCache(maxsize, typed)
