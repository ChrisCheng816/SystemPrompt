def absorb(args):    """    Given an `args` sequence of expressions, return a new list of expression    applying absorption and negative absorption.    See https://en.wikipedia.org/wiki/Absorption_law    Absorption::        A & (A | B) = A, A | (A & B) = A    Negative absorption::        A & (~A | B) = A & B, A | (~A & B) = A | B    """    def apply_absorption(expr):        if isinstance(expr, tuple) and len(expr) == 3 and expr[0] in ('&', '|'):            op, a, b = expr            if op == '&':                if a == b:                    return a                if isinstance(b, tuple) and b[0] == '|':                    if b[1] == a or b[2] == a:                        return a                if isinstance(a, tuple) and a[0] == '|':                    if a[1] == a or a[2] == a:                        return a            elif op == '|':                if a == b:                    return a                if isinstance(b, tuple) and b[0] == '&':                    if b[1] == a or b[2] == a:                        return a                if isinstance(a, tuple) and a[0] == '&':                    if a[1] == a or a[2] == a:                        return a        return expr    return [apply_absorption(expr) for expr in args]

class Histogram:    def __init__(self, data):        self.data = data        self.scale = None    def scale(self, other=None, recompute=False):        """Compute or set scale (integral of the histogram).        If *other* is ``None``, return scale of this histogram. class Graph:    def __init__(self, coordinates):        self.coordinates = coordinates        self.scale = 1.0    def scale(self, other=None):        """Get or set the scale of the graph.
